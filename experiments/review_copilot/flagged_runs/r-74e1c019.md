# r-74e1c019

## タスク

商品カテゴリごとの消費税計算を追加する。

- 新規モジュール `shoplib/tax.py`
- `TaxBreakdown`（dataclass）: `subtotal: Decimal`, `tax: Decimal`, `total: Decimal`
- `calculate_tax(items: list[CartItem], rates: dict[str, Decimal], default_rate: Decimal = Decimal("0.10")) -> TaxBreakdown`

R1: 税額は明細（CartItem）ごとに「明細の小計 × 税率」を計算し、1 円未満を切り捨て（ROUND_DOWN）てから合計する
R2: 税率は商品の category をキーに rates から引く。無ければ default_rate を使う
R3: total は subtotal + tax
R4: rates の値または default_rate が 0 未満または 1 より大きければ ValueError を送出する
R5: items が空なら subtotal・tax・total はすべて 0
完了条件: 既存テストと追加したテストがすべて通ること

+from dataclasses import dataclass
+from decimal import ROUND_DOWN, Decimal
+
+from shoplib.models import CartItem
+
+_MIN_RATE = Decimal("0")
+_MAX_RATE = Decimal("1")
+_YEN = Decimal("1")
+
+
+@dataclass(frozen=True)
+class TaxBreakdown:
+    subtotal: Decimal
+    tax: Decimal
+    total: Decimal
+
+
+def _validate_rate(rate: Decimal, name: str) -> None:
+    if not _MIN_RATE <= rate <= _MAX_RATE:
+        raise ValueError(f"{name} must be between 0 and 1: {rate}")
+
+
+def calculate_tax(
+    items: list[CartItem],
+    rates: dict[str, Decimal],
+    default_rate: Decimal = Decimal("0.10"),
+) -> TaxBreakdown:
+    _validate_rate(default_rate, "default_rate")
+    for category, rate in rates.items():
+        _validate_rate(rate, f"rate for category {category!r}")
+
+    subtotal = Decimal("0")
+    tax = Decimal("0")
+    for item in items:
+        rate = rates.get(item.product.category, default_rate)
+        line_subtotal = item.subtotal
+        subtotal += line_subtotal
+        tax += (line_subtotal * rate).quantize(_YEN, rounding=ROUND_DOWN)
+    return TaxBreakdown(subtotal=subtotal, tax=tax, total=subtotal + tax)

+from decimal import Decimal

+import pytest

+from shoplib.models import CartItem, Product

+FOOD = Product("BREAD", "Bread", Decimal("105"), category="food")
+PEN = Product("PEN", "Pen", Decimal("119"))
+RATES = {"food": Decimal("0.08"), "general": Decimal("0.10")}
+
+
+def test_tax_rounded_down_per_line_then_summed():
+    # 105*0.08=8.4→8, 119*0.10=11.9→11 ; total-based would be 20.3→20
+    items = [CartItem(FOOD, 1), CartItem(PEN, 1)]
+    result = calculate_tax(items, RATES)
+    assert result == TaxBreakdown(Decimal("224"), Decimal("19"), Decimal("243"))
+
+
+def test_rounding_applies_to_line_subtotal_not_unit_price():
+    # 105*3=315, 315*0.08=25.2→25 (per-unit rounding would give 24)
+    result = calculate_tax([CartItem(FOOD, 3)], RATES)
+    assert result.tax == Decimal("25")
+
+
+def test_unknown_category_uses_default_rate():
+    item = CartItem(Product("X", "X", Decimal("1000"), category="unknown"), 1)
+    assert calculate_tax([item], RATES).tax == Decimal("100")
+    assert calculate_tax([item], RATES, default_rate=Decimal("0.05")).tax == Decimal("50")
+
+
+def test_total_is_subtotal_plus_tax():
+    result = calculate_tax([CartItem(PEN, 2)], {})
+    assert result.subtotal == Decimal("238")
+    assert result.tax == Decimal("23")
+    assert result.total == result.subtotal + result.tax
