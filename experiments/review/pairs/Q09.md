# Q09: どちらをマージすべきか

## タスク

ユーザーごとの API 呼び出しを制限するレートリミッターを追加する。

- 新規モジュール `shoplib/ratelimit.py`
- `RateLimiter(capacity: int, refill_per_sec: float, clock=time.monotonic)`
- `allow(key: str) -> bool`

R1: トークンバケット方式。キーごとに満杯（capacity 個）の状態から始まり、allow が True を返すたびにトークンを 1 消費する。トークンが 1 未満なら False を返す
R2: トークンは経過時間 × refill_per_sec だけ連続的に補充され、capacity を超えない
R3: キーごとに独立して管理する
R4: capacity が 1 未満、または refill_per_sec が 0 以下なら ValueError を送出する
R5: 複数スレッドから同時に呼ばれても、許可回数が capacity を超えない（スレッドセーフ）
R6: 時刻は clock から取得し、テストで差し替えられるようにする
完了条件: 既存テストと追加したテストがすべて通ること

+import math
+import threading
+import time
+from collections.abc import Callable
+
+
+class RateLimiter:
+    """キーごとのトークンバケット方式レートリミッター。
+
+    各キーは capacity 個のトークンが入った満杯のバケットから始まり、
+    トークンは refill_per_sec の速度で連続的に補充される（上限 capacity）。
+    スレッドセーフ。
+    """
+
+    def __init__(self, capacity: int, refill_per_sec: float, clock: Callable[[], float] = time.monotonic) -> None:
+        if capacity < 1:
+            raise ValueError("capacity must be >= 1")
+        if not refill_per_sec > 0 or math.isinf(refill_per_sec):
+            raise ValueError("refill_per_sec must be a positive finite number")
+        self._capacity = capacity
+        self._refill_per_sec = refill_per_sec
+        self._clock = clock
+        self._lock = threading.Lock()
+        # key -> (tokens, last_update)
+        self._buckets: dict[str, tuple[float, float]] = {}
+
+    def allow(self, key: str) -> bool:
+        """トークンを 1 つ消費できれば True、できなければ False を返す。"""
+        with self._lock:
+            now = self._clock()
+            if key in self._buckets:
+                tokens, last = self._buckets[key]
+                elapsed = max(0.0, now - last)  # 時計の逆行は無視する
+                tokens = min(float(self._capacity), tokens + elapsed * self._refill_per_sec)
+            else:
+                tokens = float(self._capacity)
+            allowed = tokens >= 1
+            if allowed:
+                tokens -= 1
+            self._buckets[key] = (tokens, now)
+            return allowed

+"""ユーザー（キー）ごとの API 呼び出しを制限するトークンバケット方式のレートリミッター。"""
+
+import threading
+import time
+from collections.abc import Callable
+
+
+class RateLimiter:
+    """キーごとに独立したトークンバケットで呼び出しを制限する。
+
+    - 各キーは満杯（``capacity`` 個）の状態から始まる。
+    - ``allow`` が True を返すたびにトークンを 1 消費する。トークンが 1 未満なら False。
+    - トークンは経過時間 × ``refill_per_sec`` だけ連続的に補充され、``capacity`` を超えない。
+    - 時刻は ``clock`` から取得する（テストで差し替え可能）。
+    - スレッドセーフ。
+    """
+
+    def __init__(
+        self,
+        capacity: int,
+        refill_per_sec: float,
+        clock: Callable[[], float] = time.monotonic,
+    ) -> None:
+        if capacity < 1:
+            raise ValueError(f"capacity must be >= 1, got {capacity!r}")
+        # `not x > 0` とすることで NaN も拒否する
+        if not refill_per_sec > 0:
+            raise ValueError(f"refill_per_sec must be > 0, got {refill_per_sec!r}")
+        self._capacity = capacity
+        self._refill_per_sec = refill_per_sec
+        self._clock = clock
+        # key -> (残りトークン数, 最終更新時刻)
+        self._buckets: dict[str, tuple[float, float]] = {}
+        self._lock = threading.Lock()
+
+    @property
+    def capacity(self) -> int:
+        return self._capacity
+
+    @property
+    def refill_per_sec(self) -> float:
+        return self._refill_per_sec
+
+    def allow(self, key: str) -> bool:
+        """``key`` の呼び出しを許可するなら True を返し、トークンを 1 消費する。"""
+        with self._lock:
+            now = self._clock()
+            bucket = self._buckets.get(key)
+            if bucket is None:
+                tokens = float(self._capacity)
+            else:
+                tokens, last = bucket
+                # 時計が巻き戻った場合は補充しない（トークンを減らしもしない）
+                elapsed = max(0.0, now - last)
+                tokens = min(float(self._capacity), tokens + elapsed * self._refill_per_sec)
+
+            if tokens >= 1:
+                self._buckets[key] = (tokens - 1, now)
+                return True
+            self._buckets[key] = (tokens, now)
+            return False

+import math
+import threading
+
+import pytest
+
+from shoplib.ratelimit import RateLimiter
+
+
+class FakeClock:
+    def __init__(self, now: float = 1000.0) -> None:
+        self.now = now
+
+    def __call__(self) -> float:
+        return self.now
+
+    def advance(self, seconds: float) -> None:
+        self.now += seconds
+
+
+def drain(limiter: RateLimiter, key: str) -> int:
+    """許可されなくなるまで allow を呼び、許可回数を返す。"""
+    count = 0
+    while limiter.allow(key):
+        count += 1
+        assert count <= 10_000, "limiter never denied"
+    return count
+

+    clock = FakeClock()
+    limiter = RateLimiter(3, 1.0, clock=clock)
+    assert [limiter.allow("u") for _ in range(4)] == [True, True, True, False]

+def test_denied_call_does_not_consume_token():
+    clock = FakeClock()
+    limiter = RateLimiter(1, 1.0, clock=clock)
+    assert limiter.allow("u") is True
+    for _ in range(5):
+        assert limiter.allow("u") is False
+    clock.advance(1.0)
+    assert limiter.allow("u") is True

+# R5: スレッドセーフ
+def test_concurrent_calls_never_exceed_capacity():
+    clock = FakeClock()  # 時刻固定 → 補充なし。許可は厳密に capacity 回のはず
+    capacity = 50
+    limiter = RateLimiter(capacity, 1.0, clock=clock)
+    n_threads, calls_per_thread = 16, 100
+    barrier = threading.Barrier(n_threads)
+    results: list[int] = []
+    results_lock = threading.Lock()
+
+    def worker():
+        barrier.wait()
+        allowed = sum(limiter.allow("shared") for _ in range(calls_per_thread))
+        with results_lock:
+            results.append(allowed)
+
+    threads = [threading.Thread(target=worker) for _ in range(n_threads)]
+    for t in threads:
+        t.start()
+    for t in threads:
+        t.join()
+
+    assert len(results) == n_threads
+    assert sum(results) == capacity
+
+
+def test_concurrent_calls_with_real_clock_never_exceed_capacity():
+    # 補充が極めて遅い実時計でも、許可数は capacity を超えない
+    capacity = 20
+    limiter = RateLimiter(capacity, 1e-6)
+    n_threads = 8
+    barrier = threading.Barrier(n_threads)
+    allowed = []
+    lock = threading.Lock()
+
+    def worker():
+        barrier.wait()
+        n = sum(limiter.allow("k") for _ in range(200))
+        with lock:
+            allowed.append(n)
+
+    threads = [threading.Thread(target=worker) for _ in range(n_threads)]
+    for t in threads:
+        t.start()
+    for t in threads:
+        t.join()
+    assert sum(allowed) == capacity

+def test_uses_injected_clock():

+    def clock():

+        calls += 1

+    limiter = RateLimiter(1, 1.0, clock=clock)
+    limiter.allow("u")
+    assert calls >= 1
